Feature Engineering Pipeline

Date decomposition played a decisive role: extracting saleYear, saleMonth, saleDay, saleDayOfWeek, and saleDayOfYear uncovered seasonal liquidity patterns in the construction market. 

def add_date_features(df, date_col):
    df["saleYear"] = df[date_col].dt.year
    df["saleMonth"] = df[date_col].dt.month
    df["saleDay"] = df[date_col].dt.day
    df["saleDayOfWeek"] = df[date_col].dt.dayofweek
    df["saleDayOfYear"] = df[date_col].dt.dayofyear
    df.drop(date_col, axis=1, inplace=True)
    return df
